package main

import (
	"device/arm"
	"math"
	"unsafe"

	"github.com/USA-RedDragon/dmrgo/v2/rf"
)

var (
	doRun  = "1"
	rateS  = "9600"
	inName = "in.bin\x00"
	hash   uint32 = 2166136261
	nb     uint32
	depth  uintptr
	fd     uintptr
	pb     [3]uintptr
	ob     [12]byte
	sink   int32
)

func mix(b byte) { hash = (hash ^ uint32(b)) * 16777619 }

func mix64(v uint64) {
	for range 8 {
		mix(byte(v))
		v >>= 8
	}
}

func onBursts(bs []rf.Burst) {
	for i := range bs {
		b := &bs[i]
		nb++
		for _, c := range b.Data {
			mix(c)
		}
		for _, c := range b.Fixed {
			mix(c)
		}
		for _, c := range b.CACH {
			mix(c)
		}
		mix64(math.Float64bits(b.Seconds))
		mix64(uint64(b.Distance))
		mix64(uint64(b.Stream))
		mix64(uint64(b.Slot))
		mix(byte(b.Sync))
		mix(byte(b.Voice))
		var f byte
		if b.HasCACH {
			f |= 1
		}
		if b.BS {
			f |= 2
		}
		if b.HasFixed {
			f |= 4
		}
		mix(f)
	}
}

func atoi(s string) int {
	n := 0
	for i := 0; i < len(s); i++ {
		n = n*10 + int(s[i]-'0')
	}
	return n
}

func open() {
	pb[0] = uintptr(unsafe.Pointer(unsafe.StringData(inName)))
	pb[1] = 1
	pb[2] = uintptr(len(inName) - 1)
	fd = uintptr(arm.SemihostingCall(0x01, uintptr(unsafe.Pointer(&pb))))
}

func read(p unsafe.Pointer, n int) int {
	pb[0], pb[1], pb[2] = fd, uintptr(p), uintptr(n)
	return n - arm.SemihostingCall(0x06, uintptr(unsafe.Pointer(&pb)))
}

func paint() uintptr {
	sp := arm.AsmFull("mov {}, sp", nil)
	for a := sp - 3072; a < sp-64; a += 4 {
		*(*uint32)(unsafe.Pointer(a)) = 0xa5a5a5a5
	}
	return sp
}

func used(sp uintptr) uintptr {
	for a := sp - 3072; a < sp-64; a += 4 {
		if *(*uint32)(unsafe.Pointer(a)) != 0xa5a5a5a5 {
			return sp - a
		}
	}
	return 0
}

func put(v uint32, base uint32) {
	i := len(ob) - 1
	ob[i] = 0
	for {
		i--
		d := v % base
		if d < 10 {
			ob[i] = byte('0' + d)
		} else {
			ob[i] = byte('a' + d - 10)
		}
		v /= base
		if v == 0 {
			break
		}
	}
	arm.SemihostingCall(0x04, uintptr(unsafe.Pointer(&ob[i])))
	arm.SemihostingCall(0x03, uintptr(unsafe.Pointer(&ob[0])))
}

func main() {
	open()
	sp := paint()
	ok := run()
	depth = used(sp)
	ob[0] = ' '
	if !ok {
		nb = 0xffffffff
	}
	put(uint32(depth), 10)
	put(nb, 10)
	put(hash^uint32(sink), 16)
	ob[0] = '\n'
	arm.SemihostingCall(0x03, uintptr(unsafe.Pointer(&ob[0])))
	arm.SemihostingCall(0x18, 0x20026)
}
